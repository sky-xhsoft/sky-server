# 云盘前端拆分方案设计

## 1. 概述

### 1.1 背景

Sky-Server 云盘服务目前采用单体架构，随着业务增长，需要对前端进行模块化拆分，以提升代码可维护性、可扩展性和开发效率。

### 1.2 目标

- 将云盘前端从单体应用中拆分为独立的可复用模块
- 支持按需加载，减少首屏加载时间
- 提供清晰的组件接口和文档
- 支持独立开发和测试
- 便于后续微前端架构迁移

### 1.3 范围

本文档设计以下云盘功能模块的拆分方案：

- 文件管理模块（文件浏览、上传、下载）
- 文件夹管理模块（创建、重命名、移动、删除）
- 文件预览模块
- 文件分享模块
- 标签页管理模块
- 存储配置管理模块

```typescript
// packages/sky-cloud-core/src/api/file.ts
import { request } from '@/utils/request'
import type {
  CloudFile,
  FileListParams,
  FileListResponse,
  UploadFileParams,
  UploadFileResponse,
  RenameFileParams,
  MoveFileParams,
  DeleteFileParams
} from '@/types'

export const fileApi = {
  /** 获取文件列表 */
  list: (params: FileListParams) => {
    return request.get<FileListResponse>('/cloud/files', { params })
  },

  /** 获取文件详情 */
  get: (fileId: number) => {
    return request.get<CloudFile>(`/cloud/files/${fileId}`)
  },

  /** 重命名文件 */
  rename: (fileId: number, params: RenameFileParams) => {
    return request.put(`/cloud/files/${fileId}/rename`, params)
  },

  /** 移动文件 */
  move: (fileId: number, params: MoveFileParams) => {
    return request.put(`/cloud/files/${fileId}/move`, params)
  },

  /** 删除文件 */
  delete: (fileId: number) => {
    return request.delete(`/cloud/files/${fileId}`)
  },

  /** 批量删除文件 */
  batchDelete: (fileIds: number[]) => {
    return request.post('/cloud/files/batch-delete', { fileIds })
  },

  /** 下载文件 */
  download: (fileId: number) => {
    return request.get(`/cloud/files/${fileId}/download`, {
      responseType: 'blob'
    })
  },

  /** 预览文件 */
  preview: (fileId: number) => {
    return request.get(`/cloud/files/${fileId}/preview`)
  }
}
```

### 4.2 状态管理（Store）设计

#### 4.2.1 文件状态管理

```typescript
// packages/sky-cloud-core/src/stores/fileStore.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { fileApi, itemApi } from '@/api'
import type { CloudFile, CloudItem, ItemListParams } from '@/types'

export const useFileStore = defineStore('cloudFile', () => {
  // 状态
  const files = ref<CloudFile[]>([])
  const folders = ref<CloudItem[]>([])
  const currentFolderId = ref<number | null>(null)
  const loading = ref(false)
  const selectedItems = ref<CloudItem[]>([])

  // 计算属性
  const hasSelected = computed(() => selectedItems.value.length > 0)
  const selectedFiles = computed(() =>
    selectedItems.value.filter(item => item.itemType === 'file')
  )
  const selectedFolders = computed(() =>
    selectedItems.value.filter(item => item.itemType === 'folder')
  )

  // 动作
  async function loadItems(params?: ItemListParams) {
    loading.value = true
    try {
      const response = await itemApi.list({
        parentId: currentFolderId.value,
        ...params
      })
      folders.value = response.data.folders
      files.value = response.data.files
    } finally {
      loading.value = false
    }
  }

  function toggleSelect(item: CloudItem) {
    const index = selectedItems.value.findIndex(i => i.id === item.id)
    if (index > -1) {
      selectedItems.value.splice(index, 1)
    } else {
      selectedItems.value.push(item)
    }
  }

  function selectAll() {
    selectedItems.value = [...folders.value, ...files.value]
  }

  function clearSelection() {
    selectedItems.value = []
  }

  function setCurrentFolder(folderId: number | null) {
    currentFolderId.value = folderId
    clearSelection()
  }

  return {
    // 状态
    files,
    folders,
    currentFolderId,
    loading,
    selectedItems,

    // 计算属性
    hasSelected,
    selectedFiles,
    selectedFolders,

    // 动作
    loadItems,
    toggleSelect,
    selectAll,
    clearSelection,
    setCurrentFolder
  }
})
```

```typescript
// packages/sky-cloud-core/src/stores/uploadStore.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { UploadItem, UploadStatus } from '@/types'

export const useUploadStore = defineStore('cloudUpload', () => {
  // 状态
  const uploadQueue = ref<UploadItem[]>([])
  const uploading = ref(false)

  // 计算属性
  const uploadingItems = computed(() =>
    uploadQueue.value.filter(item => item.status === 'uploading')
  )
  const completedItems = computed(() =>
    uploadQueue.value.filter(item => item.status === 'completed')
  )
  const failedItems = computed(() =>
    uploadQueue.value.filter(item => item.status === 'failed')
  )
  const totalProgress = computed(() => {
    if (uploadQueue.value.length === 0) return 0
    const total = uploadQueue.value.reduce(
      (sum, item) => sum + (item.progress || 0),
      0
    )
    return total / uploadQueue.value.length
  })

  // 动作
  function addToQueue(file: File, folderId?: number) {
    const item: UploadItem = {
      id: Date.now().toString(),
      file,
      name: file.name,
      size: file.size,
      progress: 0,
      status: 'pending',
      folderId
    }
    uploadQueue.value.push(item)
  }

  function updateProgress(id: string, progress: number) {
    const item = uploadQueue.value.find(i => i.id === id)
    if (item) {
      item.progress = progress
    }
  }

  function updateStatus(id: string, status: UploadStatus) {
    const item = uploadQueue.value.find(i => i.id === id)
    if (item) {
      item.status = status
    }
  }

  function removeFromQueue(id: string) {
    const index = uploadQueue.value.findIndex(i => i.id === id)
    if (index > -1) {
      uploadQueue.value.splice(index, 1)
    }
  }

  function clearCompleted() {
    uploadQueue.value = uploadQueue.value.filter(
      item => item.status !== 'completed'
    )
  }

  return {
    // 状态
    uploadQueue,
    uploading,

    // 计算属性
    uploadingItems,
    completedItems,
    failedItems,
    totalProgress,

    // 动作
    addToQueue,
    updateProgress,
    updateStatus,
    removeFromQueue,
    clearCompleted
  }
})
```

#### 4.2.3 标签页状态管理

```typescript
// packages/sky-cloud-core/src/stores/tabStore.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { CloudTab, KeepAliveInclude } from '@/types'

export const useTabStore = defineStore('cloudTab', () => {
  // 状态
  const tabs = ref<CloudTab[]>([])
  const activeTabKey = ref<string>('')
  const keepAliveIncludes = ref<KeepAliveInclude[]>([])

  // 计算属性
  const activeTab = computed(() =>
    tabs.value.find(tab => tab.key === activeTabKey.value)
  )

  // 动作
  function addTab(tab: CloudTab) {
    const existing = tabs.value.find(t => t.key === tab.key)
    if (existing) {
      activeTabKey.value = tab.key
      return
    }

    tabs.value.push(tab)
    activeTabKey.value = tab.key

    if (!tab.isFixed) {
      keepAliveIncludes.value.push(tab.key)
    }
  }

  function removeTab(key: string) {
    const index = tabs.value.findIndex(tab => tab.key === key)
    if (index > -1 && !tabs.value[index].isFixed) {
      tabs.value.splice(index, 1)

      // 更新活跃标签
      if (activeTabKey.value === key) {
        if (tabs.value.length > 0) {
          activeTabKey.value = tabs.value[Math.min(index, tabs.value.length - 1)].key
        } else {
          activeTabKey.value = ''
        }
      }

      // 移除 keep-alive 缓存
      const keepIndex = keepAliveIncludes.value.indexOf(key)
      if (keepIndex > -1) {
        keepAliveIncludes.value.splice(keepIndex, 1)
      }
    }
  }

  function switchTab(key: string) {
    if (tabs.value.some(tab => tab.key === key)) {
      activeTabKey.value = key
    }
  }

  function closeOtherTabs(key: string) {
    tabs.value = tabs.value.filter(
      tab => tab.key === key || tab.isFixed
    )
    keepAliveIncludes.value = keepAliveIncludes.value.filter(
      include => include === key || tabs.value.some(tab => tab.key === include)
    )
    activeTabKey.value = key
  }

  function closeAllTabs() {
    tabs.value = tabs.value.filter(tab => tab.isFixed)
    keepAliveIncludes.value = []
    if (tabs.value.length > 0) {
      activeTabKey.value = tabs.value[0].key
    }
  }

  return {
    // 状态
    tabs,
    activeTabKey,
    keepAliveIncludes,

    // 计算属性
    activeTab,

    // 动作
    addTab,
    removeTab,
    switchTab,
    closeOtherTabs,
    closeAllTabs
  }
})
```

#### 4.3.1 文件操作 Hook

```typescript
// packages/sky-cloud-core/src/composables/useFile.ts
import { ref } from 'vue'
import { Message } from '@arco-design/web-vue'
import { fileApi, itemApi } from '@/api'
import { useFileStore } from '@/stores'

export function useFile() {
  const fileStore = useFileStore()
  const loading = ref(false)

  /** 删除文件 */
  async function deleteFile(fileId: number, fileIds?: number[]) {
    loading.value = true
    try {
      if (fileIds && fileIds.length > 1) {
        await itemApi.batchDelete({ itemIds: fileIds })
        Message.success('批量删除成功')
      } else {
        await itemApi.delete(fileId)
        Message.success('删除成功')
      }
      await fileStore.loadItems()
    } catch (error) {
      Message.error('删除失败')
      throw error
    } finally {
      loading.value = false
    }
  }

  /** 重命名文件 */
  async function renameFile(fileId: number, newName: string) {
    loading.value = true
    try {
      await itemApi.rename(fileId, { newName })
      Message.success('重命名成功')
      await fileStore.loadItems()
    } catch (error) {
      Message.error('重命名失败')
      throw error
    } finally {
      loading.value = false
    }
  }

  /** 移动文件 */
  async function moveFile(fileId: number, targetParentId: number) {
    loading.value = true
    try {
      await itemApi.move(fileId, { targetParentId })
      Message.success('移动成功')
      await fileStore.loadItems()
    } catch (error) {
      Message.error('移动失败')
      throw error
    } finally {
      loading.value = false
    }
  }

  return {
    loading,
    deleteFile,
    renameFile,
    moveFile
  }
}
```

```typescript
import { Message } from '@arco-design/web-vue'
import { useUploadStore } from '@/stores'
import type { UploadItem } from '@/types'

export function useUpload() {
  const uploadStore = useUploadStore()
  const uploading = ref(false)

  /** 上传文件 */
  async function uploadFile(item: UploadItem) {
    uploading.value = true
    uploadStore.updateStatus(item.id, 'uploading')

    try {
      // 使用 FormData 上传
      const formData = new FormData()
      formData.append('file', item.file)
      if (item.folderId) {
        formData.append('folderId', String(item.folderId))
      }

      // 模拟上传进度（实际应使用 axios 的 onUploadProgress）
      for (let i = 0; i <= 100; i += 10) {
        await new Promise(resolve => setTimeout(resolve, 100))
        uploadStore.updateProgress(item.id, i)
      }

      uploadStore.updateStatus(item.id, 'completed')
      Message.success(`${item.name} 上传成功`)
    } catch (error) {
      uploadStore.updateStatus(item.id, 'failed')
      Message.error(`${item.name} 上传失败`)
      throw error
    } finally {
      uploading.value = false
    }
  }

  /** 批量上传 */
  async function batchUpload(items: UploadItem[]) {
    for (const item of items) {
      if (item.status === 'pending' || item.status === 'failed') {
        await uploadFile(item)
      }
    }
  }

  /** 重试上传 */
  function retryUpload(item: UploadItem) {
    uploadStore.updateProgress(item.id, 0)
    uploadStore.updateStatus(item.id, 'pending')
    return uploadFile(item)
  }

  return {
    uploading,
    uploadFile,
    batchUpload,
    retryUpload
  }
}
```

```vue
const expandedKeys = ref<string[]>(props.defaultExpandedKeys || [])

  key: 'id',
  title: 'name',
  children: 'children'
}

})

// 方法
function transformToTree(folders: CloudFolder[]): any[] {
  // 实现树形结构转换逻辑
  return folders.map(folder => ({
    ...folder,
    key: String(folder.id),
    isLeaf: !folder.children || folder.children.length === 0,
    children: folder.children ? transformToTree(folder.children) : undefined
  }))
}

function handleSelect(keys: string[]) {
  selectedKeys.value = keys
  if (keys.length > 0) {
    emit('select', Number(keys[0]))
  }
}

function handleExpand(keys: string[]) {
  expandedKeys.value = keys
  emit('expand', keys)
}

// 监听默认选中
watch(
  () => props.defaultSelectedKey,
  (key) => {
    if (key) {
      selectedKeys.value = [String(key)]
    }
  }
)
```
